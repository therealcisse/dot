# Triage Report Format

Defines the structured JSON schema for the triage report generated by the oncall-triage skill. This schema is the contract downstream skills (`oncall-collect`, hypothesis investigation) build on — change it only additively.

## When to Use

| Phase | Usage |
|-------|-------|
| Phase 6 | Generate full report (`triage.json` + rendered `triage.md`) |
| Phase 7 | Broadcast uses `triage.md` content; PD note links the bundle |
| Downstream | `oncall-collect` reads `incident_id`, `input_mode`, `mcp_coverage`; hypothesis agents read `symptoms`, `recent_changes`, `initial_hypotheses` |

## JSON Schema

```json
{
  "$schema": "http://json-schema.org/draft-07/schema#",
  "title": "Triage Report",
  "type": "object",
  "required": [
    "incident_id",
    "bundle_path",
    "generated_at",
    "input_mode",
    "severity_suspected",
    "affected_surfaces",
    "symptoms",
    "impact_started_at",
    "recent_changes",
    "initial_hypotheses",
    "evidence_confidence",
    "mcp_coverage",
    "root_cause",
    "next_action",
    "completion_status"
  ],
  "properties": {
    "incident_id": {
      "type": "string",
      "description": "PagerDuty incident ID, or synthetic adhoc-YYYYMMDD-HHMMSS-slug ID"
    },
    "bundle_path": {
      "type": "string",
      "description": "Absolute path to the evidence bundle"
    },
    "generated_at": { "type": "string", "format": "date-time" },
    "input_mode": {
      "type": "string",
      "enum": ["pd_incident", "free_text_bound", "free_text_synthetic"]
    },
    "severity_suspected": {
      "type": "string",
      "enum": ["suspected-sev-1", "suspected-sev-2", "suspected-sev-3", "suspected-sev-4", "unknown"],
      "description": "Always a suspicion, never a confirmation. PD urgency informs but does not decide."
    },
    "affected_surfaces": {
      "type": "array",
      "items": {
        "type": "object",
        "required": ["name", "source"],
        "properties": {
          "name": { "type": "string", "description": "Service, endpoint, topic, or region" },
          "source": { "type": "string", "enum": ["pagerduty", "datadog", "spinnaker", "input"] }
        }
      }
    },
    "symptoms": {
      "type": "array",
      "minItems": 1,
      "items": {
        "type": "object",
        "required": ["name", "observed_at", "source"],
        "properties": {
          "name": { "type": "string", "description": "e.g. p99 latency > 4s" },
          "value": { "type": "string" },
          "observed_at": { "type": "string", "format": "date-time" },
          "source": { "type": "string", "description": "Monitor name, alert, or user report" },
          "evidence_ref": { "type": "string", "description": "Path into the bundle" }
        }
      }
    },
    "impact_started_at": {
      "type": ["string", "null"],
      "format": "date-time",
      "description": "Earliest symptom onset; null if unknown"
    },
    "blast_radius_hint": {
      "type": ["object", "null"],
      "properties": {
        "downstream": { "type": "array", "items": { "type": "string" } },
        "expanding": { "type": ["boolean", "null"] }
      }
    },
    "recent_changes": {
      "type": "array",
      "items": {
        "type": "object",
        "required": ["type", "summary", "at", "source"],
        "properties": {
          "type": { "type": "string", "enum": ["deployment", "config", "infra", "unknown"] },
          "summary": { "type": "string", "description": "e.g. checkout-api v2.14.3 deployed to prod" },
          "at": { "type": "string", "format": "date-time" },
          "source": { "type": "string" },
          "evidence_ref": { "type": "string" }
        }
      }
    },
    "initial_hypotheses": {
      "type": "array",
      "minItems": 1,
      "maxItems": 5,
      "items": {
        "type": "object",
        "required": ["id", "label", "rationale", "evidence_refs"],
        "properties": {
          "id": { "type": "string", "description": "H1..H5" },
          "label": { "type": "string", "description": "Short candidate label, not a conclusion" },
          "rationale": { "type": "string", "description": "One sentence citing evidence; correlation wording only" },
          "evidence_refs": { "type": "array", "items": { "type": "string" }, "minItems": 1 }
        }
      }
    },
    "evidence_confidence": {
      "type": "string",
      "enum": ["low", "medium", "high"],
      "description": "Confidence in the evidence snapshot, not in any hypothesis"
    },
    "mcp_coverage": {
      "type": "object",
      "required": ["pagerduty", "datadog", "spinnaker", "aws", "teams"],
      "properties": {
        "pagerduty": { "type": "string", "enum": ["ok", "degraded", "unavailable"] },
        "datadog": { "type": "string", "enum": ["ok", "degraded", "unavailable"] },
        "spinnaker": { "type": "string", "enum": ["ok", "degraded", "unavailable", "unmapped"] },
        "aws": { "type": "string", "enum": ["ok", "degraded", "unavailable"] },
        "teams": { "type": "string", "enum": ["ok", "degraded", "unavailable"] }
      }
    },
    "root_cause": {
      "type": "null",
      "description": "Always null at triage. Enforced by schema."
    },
    "next_action": {
      "type": "string",
      "const": "oncall-collect",
      "description": "Triage names the next skill; it never runs it"
    },
    "completion_status": {
      "type": "string",
      "enum": ["DONE", "DONE_WITH_CONCERNS", "BLOCKED", "NEEDS_CONTEXT"]
    }
  }
}
```

## Field Rules

- `root_cause` is schema-enforced `null`. Any RCA content belongs to a later skill's report.
- `initial_hypotheses[].rationale` must use correlation wording ("X correlates with onset at T"), never causal wording ("X caused the incident").
- `severity_suspected` is the triage's own judgment from symptom evidence; it may disagree with PD urgency.
- Timestamps are ISO-8601 UTC.

## Example (abridged)

```json
{
  "incident_id": "P8XYZ12",
  "bundle_path": "/Users/therealcisse/.incidents/P8XYZ12",
  "generated_at": "2026-09-21T14:53:10Z",
  "input_mode": "pd_incident",
  "severity_suspected": "suspected-sev-2",
  "affected_surfaces": [
    { "name": "checkout-api", "source": "pagerduty" },
    { "name": "checkout-worker", "source": "datadog" }
  ],
  "symptoms": [
    { "name": "p99 latency", "value": "4.7s", "observed_at": "2026-09-21T14:37:00Z", "source": "checkout p99 monitor", "evidence_ref": "monitors/firing.json" }
  ],
  "impact_started_at": "2026-09-21T14:37:00Z",
  "recent_changes": [
    { "type": "deployment", "summary": "checkout-api v2.14.3 to prod", "at": "2026-09-21T14:31:00Z", "source": "spinnaker", "evidence_ref": "deployments/executions.json" }
  ],
  "initial_hypotheses": [
    { "id": "H1", "label": "v2.14.3 regression", "rationale": "Deploy at 14:31 correlates with symptom onset at 14:37", "evidence_refs": ["deployments/executions.json"] },
    { "id": "H2", "label": "downstream payment latency", "rationale": "No error-rate change; latency-only symptoms", "evidence_refs": ["monitors/firing.json"] }
  ],
  "evidence_confidence": "medium",
  "mcp_coverage": { "pagerduty": "ok", "datadog": "ok", "spinnaker": "ok", "aws": "ok", "teams": "ok" },
  "root_cause": null,
  "next_action": "oncall-collect",
  "completion_status": "DONE"
}
```

## Output Location

`~/.incidents/<incident-id>/triage.json` (machine) and `triage.md` (human rendering of the same data, including the literal line "No RCA yet.").
